FONCTION chercherPlace(liste,rendu)
	POUR i de 0 à taille de liste-1
		SI  liste[i] <= rendu[0] ET rendu[0] < liste[i+1] //place trouvée
		 	place<- i+1
		FIN SI
	FIN POUR
FIN FONCTION
	
	
FONCTION ranger(liste,place,n) //n<-taille de liste-1 | place<-index où ranger le livre
	SI  n = place
		RETOURNER "livre rangé"
	SINON
		temp ← liste[n] 
		liste[n] ← tableau[n - 1] 
		liste[n - 1] ← temp 
		ranger(liste,place,n-1)
	FIN SI
FIN FONCTION
