FONCTION chercherMax(liste)
	max <-liste[0][2]
	POUR i de 1 à taille de liste -1
		SI liste[i][2]> max
			max <- liste[i][2]
		FIN SI
	FIN POUR
	RETOURNER max
FIN FONCTION

FONCTION commander(liste,valeur)
	commande <- []
	POUR i de 0 à taille de liste -1
		SI liste[i][2]==valeur
			AJOUTER liste[i] à commande
		FIN SI
	FIN POUR
	RETOURNER commande
FIN FONCTION
