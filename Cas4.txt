
FONCTION triParSelection(tableau)
n ← longueur(tableau)

	POUR i de 0 à n – 2
		indexMin ← i
		POUR j de i + 1 à n – 1
			SI tableau[j][1] < tableau[indexMin][1]
				indexMin ← j
			FIN SI
		FIN POUR
		SI indexMin != i
			temp ← tableau[i]
			tableau[i] ← tableau[indexMin]
			tableau[indexMin] ← temp
		FIN SI
	FIN POUR
FIN FONCTION
