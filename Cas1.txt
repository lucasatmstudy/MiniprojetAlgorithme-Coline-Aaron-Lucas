//Fonction pour délivrer un médicament grace a une odronnance
FONCTION delivrer(listeproduit,ordonnance)
	POUR i de 0 a taille listeproduit -1
		SI listeproduit[i][0] = ordonnance[0]			//Si le produit est trouvé
			SI listeproduit[i][1] = 0			//Le stock est vide
				RETOURNER "Hors stock"
			SINON SI listeproduit[i][1] < ordonnance[1]	//Le stock est insuffisant
				RETOURNER "Quantité insuffisante"
			SINON 
				listeproduit[i][1] <- listeproduit[i][1] - ordonnance[1] //Le stock est suffisant et gestion du stock
				RETOURNER "Stock suffisant le produit est délivré"
			FIN SI
		FIN SI
	FIN POUR
	RETOURNER "Produit non disponible"
FIN FONCTION

//Algorithme principale
DEMANDER ordonnance
ordonnance <- entrée utilisateur
listeproduit <- entrée par utilisateur
resultat <- delivrer(listeproduit,ordonnance)
AFFICHER resultat
