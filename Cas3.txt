FONCTION verifStock(tableau)
	POUR i de 0 à taille tableau -1
		SI tableau[i][1] > 0
			RETOURNER i
		FIN SI
	FIN POUR
	RETOURNER -1
FIN FONCTION


ALGORITHME PRINCIPAL

enStock <- verifStock(tableau)

SI enStock = -1
	AFFICHER "Aucun produit en stock"
SINON
	min <- tableau[enStock][2] 
	POUR i de 0 à taille tableau -1
		SI tableau[i][1] > 0
			SI tableau[i][2] < min 
				min <- tableau[i][2]
				position <- i
			FIN SI
		FIN SI
	FIN POUR
	AFFICHER "Le médicament en stock le moins cher est " + tableau[position][0] + "au prix de " + min + "€"
FIN SI
