Cas 2 Alerter sur les médicaments périmés 

listeproduit <- [[Avamix, 03/04/2026], [Amoxicilline, 04/07/2026], [Ozempic, 0	05/12/2025], [Paracétamol, 13/05/2026], [Doliprane, 24/07/2027], [Spacefond, 07/04/2026], [Efferalgan, 05/09/2026]]
perime <- []
bientotperime <- []
DEMANDER date
date <- date ajoute par l'utilisateur
POUR i de 0 à taille de listeproduit -1
	SI listeproduit[i][1] < date
		AFFICHER listeproduit[i][0] + "périmé"
		AJOUTER listeproduit [i][0] à perime
	SINON SI listeproduit[i][1] < date+30jours
		AFFICHER listeproduit[i][0] + "bientot périmé"
		AJOUTER listeproduit [i][0] à bientotperime
	FIN SI
FIN POUR
