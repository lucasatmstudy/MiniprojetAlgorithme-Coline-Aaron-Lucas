//Fonction de tri
FONCTION triretard(listeretard)
	POUR i de 0 à taille listeretard -2
		indexlivre <- i
		POUR j de i+1 à taille listeretard -1
			SI listeretard [i][2] < listeretard[indexlivre][2]
				indexlivre <- j
			FIN SI
		FIN POUR
		SI indexlivre != i
			temp ← listeretard[i]
			listeretard[i] ← listeretard[indexlivre]
			listeretard[indexlivre] ← temp
		FIN SI
	FIN POUR
FIN FONCTION
